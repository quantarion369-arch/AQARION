def edit(text, old, new, label):
    count = text.count(old)
    if count != 1:
        raise RuntimeError(
            f"Anchor mismatch: {label}; expected 1, found {count}"
        )
    return text.replace(old, new, 1)

def pytest_run(arguments, filename):
    completed = subprocess.run(
        [
            sys.executable, "-B", "-m", "pytest",
            "-q", "-p", "no:cacheprovider", *arguments,
        ],
        cwd=pkg,
        env=dict(os.environ, PYTHONDONTWRITEBYTECODE="1"),
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
    )
    (work / filename).write_text(completed.stdout, encoding="utf-8")
    return completed
