function assert(value,message) {
  if (!value) throw new Error(message);

async function settle() {
  await new Promise(resolve=>setTimeout(resolve,0));
  await new Promise(resolve=>setTimeout(resolve,0));

function installFixture(initialQuery) {
  const records=[], errors=[];
  window.__race={records,errors};
  window.addEventListener("error",event=>errors.push(String(event.message)));
  window.addEventListener("unhandledrejection",
    event=>errors.push(String(event.reason)));
  document.getElementById("q").value=initialQuery;
  window.fetch=(url,options={})=>new Promise(resolve=>{
    let bodyResolve,bodyReject;
    const body=new Promise((yes,no)=>{bodyResolve=yes;bodyReject=no;});
    const request={
      url:new URL(String(url),document.baseURI),
      signal:options.signal,
      jsonStarted:false,
      headers(status=200) {
        resolve({
          ok:status>=200 && status<300,
          status,
          json:()=>{request.jsonStarted=true;return body;}
        });
      },
      body:bodyResolve,
      rejectBody:bodyReject,
      reply(data,status=200) {
        request.headers(status);
        bodyResolve(data);
      }
    };
    records.push(request);
    // Abort is deliberately ignored: stale responses can still finish.
  });
}

        passage:text

      supplementary_context:[]
    },
    related_evidence:[],
    relationship_diagnostics:[]
  };
}

class Handler(BaseHTTPRequestHandler):
    def send_bytes(self, data, content_type, status=200):
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Cache-Control", "no-store")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        path = self.path.split("?", 1)[0]
        routes = {
            "/": (PAGE.encode(), "text/html; charset=utf-8"),
            "/application": (APP, "text/html; charset=utf-8"),
            "/negative-control": (MUTANT, "text/html; charset=utf-8"),
            "/metadata": (json.dumps(META).encode(), "application/json"),
        }
        if path not in routes:
            self.send_bytes(b"Not found", "text/plain", 404)
            return
        self.send_bytes(*routes[path])

    def do_POST(self):
        if self.path != "/receipt":
            self.send_bytes(b"Not found", "text/plain", 404)
            return
        length = int(self.headers.get("Content-Length", "0"))
        if not 0 < length <= 128000:
            self.send_bytes(b"Invalid length", "text/plain", 400)
            return
        report = json.loads(self.rfile.read(length))
        if any(report.get(key) != value for key, value in META.items()):
            self.send_bytes(b"Metadata mismatch", "text/plain", 400)
            return
        report["server_received_at_utc"] = datetime.now(timezone.utc).isoformat()
        report["application_still_matches_snapshot"] = (
            (REPO / REL).read_bytes() == APP
        )
        filename = "browser-race-receipt-" + str(time.time_ns()) + ".json"
        (WORK / filename).write_text(
            json.dumps(report, indent=2) + "\n", encoding="utf-8"
        )
        print("BROWSER_RACE_RECEIPT:", WORK / filename, flush=True)
        print("REPORTED_RESULT:", report.get("overall"), flush=True)
        self.send_bytes(
            json.dumps({"receipt": filename}).encode(), "application/json"
        )

with HTTPServer(("127.0.0.1", 8766), Handler) as server:
