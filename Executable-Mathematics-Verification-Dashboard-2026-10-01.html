import AQARIONPullbackCore.QuotientInjective

namespace Aqarion

theorem finite_injective_endomap_bijective
    {Q : Type*}
    [Finite Q]
    (g : Q → Q)
    (hg : Function.Injective g) :
    Function.Bijective g := by
  letI : Fintype Q := Fintype.ofFinite Q
  exact ⟨hg, Finite.surjective_of_injective hg⟩

theorem pullbackFixed_quotientMap_bijective
    {X : Type u}
    (T : X → X)
    (E : Setoid X)
    [Finite (Quotient E)]
    (hFixed : PullbackFixed T E) :
    Function.Bijective (quotientMap T E hFixed) := by
  apply finite_injective_endomap_bijective
  exact quotientMap_injective T E hFixed

Dashboard principles

  1. A definition must be locked before code or data are trusted as evidence.
  2. Finite exhaustive computation is a scoped computational certificate, not a universal theorem.
  3. Independent agreement is stronger than self-consistency.
  4. Mutation tests show whether a validator distinguishes plausible wrong semantics.
  5. A formal proof claim requires a clean toolchain-specific build receipt.
  6. An AI agent can call tools and explain evidence, but may not silently promote claim status.
